def logical_left_shift_8bit(binary_str, shift_amt):
    # Ensure input is exactly 8 bits (taking the rightmost 8 bits if larger)
    orig_8bit = binary_str[-8:]
    print(f"Original 8-bit Input: {orig_8bit}")
    
    current = orig_8bit
    # Step-by-step breakdown
    for step in range(1, shift_amt + 1):
        # Drop the leftmost bit, shift left, and append '0' on the right
        dropped_bit = current[0]
        current = current[1:] + "0"
        print(f"  Step {step}: Shift Left by 1 -> Drops '{dropped_bit}', appends '0' -> Result: {current}")
        
    return current

def logical_right_shift_8bit(binary_str, shift_amt):
    orig_8bit = binary_str[-8:]
    print(f"Original 8-bit Input: {orig_8bit}")
    
    current = orig_8bit
    # Step-by-step breakdown
    for step in range(1, shift_amt + 1):
        # Drop the rightmost bit, shift right, and prepend '0' on the left
        dropped_bit = current[-1]
        current = "0" + current[:-1]
        print(f"  Step {step}: Shift Right by 1 -> Drops '{dropped_bit}', prepends '0' -> Result: {current}")
        
    return current

print("--- Operation A: Logical Left Shift by 2 ---")
# Using the lower 8 bits of 1111110110 -> 11110110
result_a = logical_left_shift_8bit("1111110110", 2)
print(f"Final 8-bit Result:   {result_a}\n")

print("--- Operation B: Logical Right Shift by 2 ---")
result_b = logical_right_shift_8bit("10010100", 2)
print(f"Final 8-bit Result:   {result_b}\n")
